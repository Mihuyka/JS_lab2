function longest(arr, n) {
    let answ = [];
    let l = arr.length;
    
    for (let i = 0; i < l; i++) {
        answ[i] = arr[i];
    }

    for (let i = 0; i < l - 1; i++) {
        for (let c = 0; c < l - i - 1; c++) {
            if (answ[c].length < answ[c + 1].length) {
                let t = answ[c];
                answ[c] = answ[c + 1];
                answ[c + 1] = t;
            }
        }
    }

    return answ[n - 1];
}
