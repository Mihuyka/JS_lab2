function primeFactors(n){
    let answ = "";
    let i = 2;
    
    while (i * i <= n) {
        let c = 0;
        
        while (n % i === 0) {
            c++;
            n /= i;
        }
        
        if (c === 1) {
            answ += "(" + i + ")";
        } else if (c > 1) {
            answ += "(" + i + "**" + c + ")";
        }
        
        i++;
    }
    
    if (n > 1) {
        answ += "(" + n + ")";
    }
    
    return answ;
}
