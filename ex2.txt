function ipv4Parser(ip, mask){
  let ipPieces = ip.split(".");
  let maskPieces = mask.split(".");
  let netPieces = [];
  let hostPieces = [];
  
  for (let i = 0; i < 4; ++i) {
    let ipN = Number(ipPieces[i]);
    let maskN = Number(maskPieces[i]);
    let netN = ipN & maskN;
    let hostN = ipN ^ netN;
    netPieces.push(netN);
    hostPieces.push(hostN);
  }
  
  let result = [netPieces.join("."), hostPieces.join(".")];
  return result;
}
