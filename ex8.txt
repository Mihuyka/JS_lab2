function expandedForm(num) {
  let s = String(num);
    let answ = "";
    let n = s.length;

    for (let i = 0; i < n; i++) {
        if (s[i] !== '0') {
            let t = s[i];
            
            for (let c = i + 1; c < n; ++c) {
                t += "0";
            }
            
            if (answ === "") {
                answ = t;
            } else {
                answ += " + " + t;
            }
        }
    }

    return answ;
}
